import java.util.*;

class PageReplacementMenu {
    static Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        while (true) {
            System.out.println("\n---------------- PAGE REPLACEMENT ALGORITHMS ----------------");
            System.out.println("1. First In First Out (FIFO)");
            System.out.println("2. Optimal Page Replacement");
            System.out.println("3. Least Recently Used (LRU)");
            System.out.println("4. Exit");
            System.out.print("Enter your choice: ");
            int ch = sc.nextInt();

            switch (ch) {
                case 1 -> fifo();
                case 2 -> optimal();
                case 3 -> lru();
                case 4 -> {
                    System.out.println("Exiting...");
                    sc.close();
                    return;
                }
                default -> System.out.println("Invalid choice! Try again.");
            }
        }
    }

    // ---------------- FIRST IN FIRST OUT ----------------
    static void fifo() {
        System.out.print("Enter number of pages: ");
        int n = sc.nextInt();
        int[] pages = new int[n];
        System.out.println("Enter reference string:");
        for (int i = 0; i < n; i++) pages[i] = sc.nextInt();

        System.out.print("Enter frame capacity: ");
        int cap = sc.nextInt();
        int[] frame = new int[cap];
        Arrays.fill(frame, -1);

        int index = 0, hit = 0, fault = 0;

        System.out.println("\nPage\tFrames\t\tStatus");
        for (int i = 0; i < n; i++) {
            boolean found = false;

            // Check for hit
            for (int j = 0; j < cap; j++) {
                if (frame[j] == pages[i]) {
                    found = true;
                    hit++;
                    break;
                }
            }

            // Page fault → replace oldest
            if (!found) {
                frame[index] = pages[i];
                index = (index + 1) % cap;
                fault++;
            }

            // Print frame contents
            System.out.print(pages[i] + "\t");
            for (int j = 0; j < cap; j++) {
                if (frame[j] == -1)
                    System.out.print("- ");
                else
                    System.out.print(frame[j] + " ");
            }
            System.out.println("\t" + (found ? "Hit" : "Fault"));
        }

        System.out.println("\nTotal Hits: " + hit);
        System.out.println("Total Faults: " + fault);
        System.out.printf("Hit Ratio: %.2f%%\n", (hit * 100.0 / n));
        System.out.printf("Fault Ratio: %.2f%%\n", (fault * 100.0 / n));
    }

    // ---------------- OPTIMAL PAGE REPLACEMENT ----------------
    static void optimal() {
        System.out.print("Enter number of pages: ");
        int n = sc.nextInt();
        int[] pages = new int[n];
        System.out.println("Enter reference string:");
        for (int i = 0; i < n; i++) pages[i] = sc.nextInt();

        System.out.print("Enter number of frames: ");
        int f = sc.nextInt();
        int[] frame = new int[f];
        Arrays.fill(frame, -1);
        int hit = 0, fault = 0;

        System.out.println("\nPg\tFrames\tStatus");
        for (int i = 0; i < n; i++) {
            int p = pages[i];
            int pos = -1;
            for (int j = 0; j < f; j++)
                if (frame[j] == p) pos = j;

            if (pos == -1) { // page fault
                fault++;
                int rep = -1, far = -1;
                for (int j = 0; j < f; j++) {
                    if (frame[j] == -1) {
                        rep = j;
                        break;
                    }
                    int k;
                    for (k = i + 1; k < n; k++)
                        if (frame[j] == pages[k]) break;
                    if (k > far) {
                        far = k;
                        rep = j;
                    }
                }
                frame[rep] = p;
            } else {
                hit++;
            }

            // Display frame status
            System.out.print(p + "\t");
            for (int j = 0; j < f; j++)
                System.out.print((frame[j] == -1 ? "- " : frame[j] + " "));
            System.out.println("\t" + (pos == -1 ? "Fault" : "Hit"));
        }

        System.out.printf("\nHits = %d, Faults = %d\nHit Ratio = %.2f%%\n",
                hit, fault, hit * 100.0 / n);
    }

    // ---------------- LEAST RECENTLY USED (LRU) ----------------
    static void lru() {
        System.out.print("Enter number of pages: ");
        int n = sc.nextInt();
        int[] pages = new int[n];
        System.out.println("Enter reference string:");
        for (int i = 0; i < n; i++) pages[i] = sc.nextInt();

        System.out.print("Enter frame capacity: ");
        int f = sc.nextInt();
        int[] frame = new int[f];
        Arrays.fill(frame, -1);

        ArrayList<Integer> recent = new ArrayList<>();
        int hit = 0, fault = 0;

        System.out.println("\nPg\tFrames\tStatus");
        for (int i = 0; i < n; i++) {
            int p = pages[i];
            boolean found = false;

            // Check for hit
            for (int j = 0; j < f; j++) {
                if (frame[j] == p) {
                    found = true;
                    hit++;
                    break;
                }
            }

            if (!found) {
                fault++;
                int rep = -1;
                for (int j = 0; j < f; j++) {
                    if (frame[j] == -1) {
                        rep = j;
                        break;
                    }
                }

                // All full → replace least recently used
                if (rep == -1) {
                    int lru = recent.get(0);
                    for (int j = 0; j < f; j++)
                        if (frame[j] == lru) {
                            rep = j;
                            break;
                        }
                    recent.remove(0);
                }
                frame[rep] = p;
            }

            // Update recent usage
            recent.remove((Integer) p);
            recent.add(p);

            // Display frame content
            System.out.print(p + "\t");
            for (int j = 0; j < f; j++)
                System.out.print((frame[j] == -1 ? "- " : frame[j] + " "));
            System.out.println("\t" + (found ? "Hit" : "Fault"));
        }

        System.out.printf("\nHits = %d, Faults = %d\nHit Ratio = %.2f%%\n",
                hit, fault, hit * 100.0 / n);
    }
}
